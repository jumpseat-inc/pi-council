# EV-30 Invocation-Scoped Spend Record — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A pure accounting module `extensions/spend.ts` that, given the invoking session's entries and the invocation's run manifests, returns a `SpendRecord` (own-session half + subtree half, each labelled with its basis) plus the one call-site stamp in `extensions/index.ts` that makes the invocation boundary durable.

**Architecture:** New pure module `extensions/spend.ts` (no `node:fs`, no ctx, no hub; types only) exporting `SpendRecord`/`SpendBoundary`, `spendRecord()`, `formatBoundaryLabel()`, `recordInvocationBoundary()`, and the marker-recovery helper. The procedure command handler in `extensions/index.ts` stamps a synchronous `council-invocation` custom entry before `sendUserMessage` in both TUI and headless branches. Boundary resolution is **append-order-based** (Skeptic O1 `closed-red`): the first on-branch user message after the marker in `getEntries()` order — never `parentId === markerId`.

**Tech Stack:** TypeScript (strict), bun:test, pi's `SessionEntry`/`parseSessionEntries`/`SessionManager.inMemory`, `RunManifest`/`Usage` from `extensions/runs.ts`.

**Spec:** `docs/superpowers/specs/2026-09-11-EV-30-design.md` (binding; step-6 product-owner ruling in `council/cards/EV-30.md`).

## Global Constraints

- **Ruling item 2:** no `total` field on the record; no new `UsageSource` union member. `UsageSource` stays `"stream-assistant" | "session-reconciled"`.
- **Ruling item 3:** `boundary: { resolved: boolean }`; unresolved label is byte-exact `boundary=session=<id> entries=unresolved jobs=0`.
- **Ruling item 4:** "rotated session" = `boundary-unresolvable`; zero **both** halves; the word "rotated" never appears in tests/prose.
- **Ruling item 5:** no extra lower-bound string; the subtree's `usageSource: "stream-assistant"` is the statement.
- **Ruling item 6:** no `computedAt`/idempotence stamp — record valid only when written once at exit.
- **Ruling item 7:** `lastId` = last entry id on the active leaf chain at compute time (= `leafId`), never "last file entry".
- **R-4 (resolved label):** byte-exact `boundary=session=<sessionId> entries=<firstId>..<lastId> jobs=<n>`.
- Own half enumerates **four kinds**: assistant `message.usage` (+ `turns`), toolResult `message.usage`, compaction `entry.usage`, `branch_summary` `entry.usage`. Walk the **raw** active chain (not `buildContextEntries`).
- Wire→record flattening (EV-28 Q1 verbatim): `cost.{input,output,cacheRead,cacheWrite,total} → costInput/costOutput/costCacheRead/costCacheWrite/cost`; missing components coerce to 0, never `NaN`; `cacheRead` never folded into `input`; `reasoning` accumulated independently.
- Subtree: eligible roots = manifests with `parentJobId === null && startedAt >= Date.parse(boundaryEntry.timestamp)`; forest = each root closed over `parentJobId` descendants; `jobCount` = all forest nodes; manifest without `usage` counts in `jobCount`, adds 0.
- Untouched files: `extensions/runs.ts`, `extensions/hub.ts`, `extensions/transcript.ts`, `extensions/navigator.ts`, seats/procedures payload, scaffold.
- Every bash call carries an explicit timeout; gates in order: `bash council/preflight.sh EV-30`, `bunx tsc --noEmit`, `bun test`, `python3 council/validate.py`.
- Conventional Commits; commits only inside the worktree `/tmp/ev30-spend`.

```ts
// extensions/spend.ts — EV-30: invocation-scoped spend record. Pure: no node:fs, no ctx, no hub.
import type { SessionEntry } from "@earendil-works/pi-coding-agent";
import type { CostBasis, RunManifest, Usage, UsageMetric, UsageSource } from "./runs.ts";

export interface SpendBoundary {
	sessionId: string;
	resolved: boolean;
	firstEntryId: string | null;
	lastEntryId: string | null;
	jobCount: number;
}

export interface SpendRecord {
	boundary: SpendBoundary;
	ownSession: Usage;  // usageSource: "session-reconciled"
	subtree: Usage;     // usageSource: "stream-assistant" — the lower-bound statement (ruling item 5)
}

/** Wire shape of pi-ai Usage as it appears on session entries (cost nested). */
interface WireUsage {
	input?: number;
	output?: number;
	cacheRead?: number;
	cacheWrite?: number;
	reasoning?: number;
	totalTokens?: number;
	cost?: { input?: number; output?: number; cacheRead?: number; cacheWrite?: number; total?: number };
}

export function zeroUsage(costBasis: CostBasis, usageSource: UsageSource): Usage {
	return {
		input: 0, output: 0, cacheRead: 0, cacheWrite: 0, reasoning: 0, totalTokens: 0,
		cost: 0, costInput: 0, costOutput: 0, costCacheRead: 0, costCacheWrite: 0,
		turns: 0, costBasis, usageSource,
	};
}

/** EV-28 Q1 wire→record clause verbatim: missing components coerce to 0, never NaN. */
function accumulateWire(into: Usage, u: WireUsage | undefined): void {
	if (!u) return;
	into.input += u.input || 0;
	into.output += u.output || 0;
	into.cacheRead += u.cacheRead || 0; // accumulated, never folded into input
	into.cacheWrite += u.cacheWrite || 0;
	into.reasoning += u.reasoning || 0; // subset of output — accumulate independently, never re-add
	into.totalTokens += u.totalTokens || 0;
	into.cost += u.cost?.total || 0;
	into.costInput += u.cost?.input || 0;
	into.costOutput += u.cost?.output || 0;
	into.costCacheRead += u.cost?.cacheRead || 0;
	into.costCacheWrite += u.cost?.cacheWrite || 0;
}

/** Flat RunManifest.usage summation (already EV-28-shaped; nested cost absent). */
const NUMERIC_METRICS: UsageMetric[] = [
	"input", "output", "cacheRead", "cacheWrite", "reasoning", "totalTokens",
	"cost", "costInput", "costOutput", "costCacheRead", "costCacheWrite", "turns",
];
function accumulateFlat(into: Usage, u: Usage | undefined): void {
	if (!u) return; // legacy/partial manifest → counts in jobCount, adds 0
	for (const k of NUMERIC_METRICS) into[k] += u[k] || 0;
}

/** Active leaf chain (root → leaf) as ids, plus the on-chain id set. */
function activeChain(entries: SessionEntry[], leafId: string | null): { ids: string[]; onChain: Set<string> } {
	const byId = new Map(entries.map((e) => [e.id, e]));
	const ids: string[] = [];
	let cur = leafId;
	while (cur) {
		const e = byId.get(cur);
		if (!e) break;
		ids.push(cur);
		cur = e.parentId;
	}
	return { ids, onChain: new Set(ids) };
}

/** Append-order scan: the last council-invocation marker on the active chain (EV-31/EV-32 interface completion). */
export function findLatestInvocationMarker(entries: SessionEntry[], leafId: string | null): string | null {
	const { onChain } = activeChain(entries, leafId);
	let found: string | null = null;
	for (const e of entries) {
		if (e.type === "custom" && e.customType === "council-invocation" && onChain.has(e.id)) found = e.id;
	}
	return found;
}

export function spendRecord(opts: {
	entries: SessionEntry[];
	leafId: string | null;
	sessionId: string;
	markerId: string | null;
	manifests: RunManifest[];
}): SpendRecord {
	const { entries, leafId, sessionId, markerId, manifests } = opts;
	const ownSession = zeroUsage("catalogue-estimate", "session-reconciled");
	const subtree = zeroUsage("catalogue-estimate", "stream-assistant");
	const byId = new Map(entries.map((e) => [e.id, e]));
	const { ids, onChain } = activeChain(entries, leafId);

	// Resolution (spec §2.5): the marker must exist and be on the active chain;
	// the boundary is the first on-chain user message after the marker in
	// getEntries() append order — never parentId === markerId (Skeptic O1).
	let boundaryId: string | null = null;
	if (markerId && byId.has(markerId) && onChain.has(markerId)) {
		const markerIdx = entries.findIndex((e) => e.id === markerId);
		for (let i = markerIdx + 1; i < entries.length; i++) {
			const e = entries[i]!;
			if (e.type === "message" && e.message.role === "user" && onChain.has(e.id)) {
				boundaryId = e.id;
				break;
			}
		}
	}
	if (!boundaryId) {
		// Boundary unresolvable (ruling items 3+4): zero BOTH halves, never an error.
		return {
			boundary: { sessionId, resolved: false, firstEntryId: null, lastEntryId: null, jobCount: 0 },
			ownSession,
			subtree,
		};
	}

	// Own half: raw active chain at/after the boundary (compacted-away entries included; NOT buildContextEntries).
	let seenBoundary = false;
	for (let i = ids.length - 1; i >= 0; i--) { // root → leaf
		const e = byId.get(ids[i]!)!;
		if (!seenBoundary) {
			if (e.id !== boundaryId) continue;
			seenBoundary = true;
		}
		if (e.type === "message") {
			if (e.message.role === "assistant") {
				accumulateWire(ownSession, e.message.usage);
				ownSession.turns += 1; // assistant-only, mirrors hub.ts
			} else if (e.message.role === "toolResult") {
				accumulateWire(ownSession, e.message.usage);
			}
		} else if (e.type === "compaction" || e.type === "branch_summary") {
			accumulateWire(ownSession, e.usage); // ruling item 1: branch_summary included
		}
	}

	// Subtree half: forest of time-eligible top-level roots closed over parentJobId descendants.
	const boundaryEntry = byId.get(boundaryId)!;
	const boundaryMs = Date.parse(boundaryEntry.timestamp);
	const byParent = new Map<string | null, RunManifest[]>();
	for (const m of manifests) {
		const list = byParent.get(m.parentJobId) ?? [];
		list.push(m);
		byParent.set(m.parentJobId, list);
	}
	const forest: RunManifest[] = [];
	const walk = (id: string): void => {
		for (const k of byParent.get(id) ?? []) {
			forest.push(k);
			walk(k.id);
		}
	};
	for (const r of byParent.get(null) ?? []) {
		if (r.startedAt >= boundaryMs) {
			forest.push(r);
			walk(r.id);
		}
	}
	for (const m of forest) accumulateFlat(subtree, m.usage);

	return {
		boundary: {
			sessionId,
			resolved: true,
			firstEntryId: boundaryId,
			lastEntryId: leafId!, // ruling item 7: last id on the active chain = leafId, never "last file entry"
			jobCount: forest.length,
		},
		ownSession,
		subtree,
	};
}

/** R-4 resolved label byte-exact; unresolved form per ruling item 3. */
export function formatBoundaryLabel(record: SpendRecord): string {
	const b = record.boundary;
	if (!b.resolved || b.firstEntryId === null || b.lastEntryId === null) {
		return `boundary=session=${b.sessionId} entries=unresolved jobs=0`;
	}
	return `boundary=session=${b.sessionId} entries=${b.firstEntryId}..${b.lastEntryId} jobs=${b.jobCount}`;
}
```

Note: `Date.parse` of an unparseable timestamp yields `NaN`, which excludes every root (zero subtree, no throw) — timestamps are ISO strings by construction (`session-manager.js:792`).

```ts
/** Stamp the invocation boundary (sync, before sendUserMessage) and return the
 * marker id. Fail closed (spec §2.4 Held #7): if the post-append leaf is not
 * the council-invocation marker (stale/replaced session), return null and
 * write no further state — an unresolved boundary is better than a wrong one. */
export function recordInvocationBoundary(
	pi: { appendEntry<T = unknown>(customType: string, data?: T): void },
	ctx: { sessionManager: { getLeafId(): string | null; getEntry(id: string): SessionEntry | undefined } },
	command: string,
	runId: string,
): string | null {
	pi.appendEntry("council-invocation", { command, runId, at: Date.now() });
	const leafId = ctx.sessionManager.getLeafId(); // pi.appendEntry returns void (O2) — the id is observable only here
	if (leafId === null) return null;
	const entry = ctx.sessionManager.getEntry(leafId);
	if (!entry || entry.type !== "custom" || entry.customType !== "council-invocation") return null;
	return leafId;
}
```
