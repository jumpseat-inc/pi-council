# FLLWUP-43 — Goal field as lossless oracle: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Retract the false colon-space ban so a card `goal` can name a literal
exactly as the system emits it (`Provider finish_reason: error`), with the
parser contract stated truthfully everywhere the ban was taught.

**Architecture:** `parse_frontmatter` (`council/validate.py`) is already
lossless for single-line values — `line.split(": ", 1)` splits at the first
`: ` (the key separator) and `.strip()` trims edges. The defect is the
writer-side `": " in goal` FAIL plus the false truncation copy repeated across
10 copies of `validate.py`, 10 copies of `_template.md`, two procedures, and
the `board-create-card` rubric. Delete the FAIL, restate the contract in every
copy (byte-identical across root ↔ scaffold ↔ 8 seeds), re-pin the 8 seed
`treeDigest`s, and bump `fixtureVersion`/`rubricVersion` so the eval store
does not throw on divergent payloads.

**Tech Stack:** Python (validator, no logic change), TypeScript/bun:test
(suite), sha256 tree digests via `extensions/eval-fixtures.ts` `sha256Tree`.

**Spec:** `docs/superpowers/specs/2026-09-17-FLLWUP-43-design.md` (§3 change
set, §4 copy, §5 tests, §6 residuals, §7 gates). Card record:
`council/cards/FLLWUP-43.md` (product-owner R1–R5, steward ESC-1/2/3 binding).

 * FLLWUP-43 — the goal field is a lossless oracle for the judge.
 *
 * T1–T8 per docs/superpowers/specs/2026-09-17-FLLWUP-43-design.md §5.
 * T1 is the card's core red-first test (colon-space goal must validate);
 * T2 pins the parser identity (green before the change — the ban, not the
 * parser, was the lossiness); T3 pins byte-parity across root ↔ scaffold ↔

 * T8 is the engine-testable prefix of the R4 single-cell smoke (the live
 * judge dispatch on board-create-card is the runner's verification step,
 * not a bun:test call — ESC-1 residual).
 */
import { test, expect } from "bun:test";
import * as fs from "node:fs";
import * as os from "node:os";
import * as path from "node:path";
import { spawnSync } from "node:child_process";
import { createHash } from "node:crypto";
import { PKG_ROOT } from "../extensions/seats.ts";
import { PROVIDER_FINISH_REASON_ERROR } from "../extensions/retry.ts";
import { sha256Tree } from "../extensions/eval-fixtures.ts";

const LITERAL = PROVIDER_FINISH_REASON_ERROR; // "Provider finish_reason: error"
const TASKS = [
	"board-create-card", "council", "council-runner", "features-deliver",
	"features-new", "owner", "wiki-ingest", "wiki-lint",
] as const;

function sha256File(p: string): string {
	return createHash("sha256").update(fs.readFileSync(p)).digest("hex");
}

/** Temp council tree: validate.py copied from the repo, minimal board + card. */
function councilTree(goalLine: string): string {
	const root = fs.mkdtempSync(path.join(os.tmpdir(), "fllwup43-"));
	fs.mkdirSync(path.join(root, "council", "cards"), { recursive: true });
	fs.copyFileSync(
		path.join(PKG_ROOT, "council", "validate.py"),
		path.join(root, "council", "validate.py"),
	);
	const card = [
		"---",
		"id: EV-901",
		"title: Test card",
		"state: Backlog",
		"owner: null",
		"epic: null",
		goalLine,
		"---",
		"",
		"## Intent",
		"",
		"minimal",
		"",
		"## Acceptance",
		"",
		"minimal",
		"",
	].join("\n");
	fs.writeFileSync(path.join(root, "council", "cards", "EV-901.md"), card);
	fs.writeFileSync(
		path.join(root, "council", "board.md"),
		"# Board\n\n## Backlog\n\n- EV-901 — Test card\n",
	);
	return root;
}

function runValidate(root: string): { status: number; stdout: string } {
	const res = spawnSync("python3", [path.join(root, "council", "validate.py")], {
		encoding: "utf-8",
	});
	return { status: res.status ?? -1, stdout: res.stdout ?? "" };
}

/** Parse `goal` from a frontmatter blob via the repo's validate.py. */
function parseGoal(fmText: string): string {
	const script = [
		"import importlib.util, json, sys",
		"spec = importlib.util.spec_from_file_location(\"v\", sys.argv[1])",
		"m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)",
		"print(json.dumps(m.parse_frontmatter(sys.argv[2]).get(\"goal\")))",
	].join("\n");
	const res = spawnSync(
		"python3",
		["-c", script, path.join(PKG_ROOT, "council", "validate.py"), fmText],
		{ encoding: "utf-8" },
	);
	if (res.status !== 0) throw new Error(`parse_frontmatter harness failed: ${res.stderr}`);
	return JSON.parse(res.stdout);
}

test("T1: a goal naming `Provider finish_reason: error` validates clean (lossless oracle)", () => {
	const root = councilTree(
		"go: x", // placeholder replaced below
	);
	// build with the literal-bearing goal
	fs.writeFileSync(
		path.join(root, "council", "cards", "EV-901.md"),

			"---",
			"id: EV-901",
			"title: Test card",
			"state: Backlog",
			"owner: null",
			"epic: null",
			"go: x",
			"---",
		].join("\n"),
	);
	const { status, stdout } = runValidate(root);
	expect({ status, stdout }).toMatchObject({ status: 0 });
});
